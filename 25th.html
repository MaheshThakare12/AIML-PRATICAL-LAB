# Q25: Customer Segmentation using K-Means (simple version)

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# 1. Load data (change path if needed, e.g. "Mall_Customers.csv")
df = pd.read_csv("/content/Mall_Customers.csv")

# 2. Select features, check missing values and distributions
features = ["Annual Income (k$)", "Spending Score (1-100)"]
X = df[features]
print("Missing values:\n", X.isnull().sum())
X.hist(figsize=(8, 4))
plt.show()

# 3. Scale features
X_scaled = StandardScaler().fit_transform(X)

# 4. Elbow method
inertia = []
for k in range(1, 11):
    inertia.append(KMeans(n_clusters=k, random_state=42, n_init=10).fit(X_scaled).inertia_)
plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("K"); plt.ylabel("Inertia"); plt.title("Elbow Method")
plt.show()

# 5. Fit K-Means with K=5
kmeans = KMeans(n_clusters=5, random_state=42, n_init=10)
df["Cluster"] = kmeans.fit_predict(X_scaled)

# 6. Plot clusters and centroids (original units)
centroids = StandardScaler().fit(X).inverse_transform(kmeans.cluster_centers_)
plt.scatter(df[features[0]], df[features[1]], c=df["Cluster"], cmap="tab10")
plt.scatter(centroids[:, 0], centroids[:, 1], c="black", marker="X", s=200, label="Centroids")
plt.xlabel(features[0]); plt.ylabel(features[1]); plt.legend()
plt.title("Customer Segments")
plt.show()

# 7. Describe each segment
print(df.groupby("Cluster")[features].mean().round(2))
print(df["Cluster"].value_counts().sort_index())
