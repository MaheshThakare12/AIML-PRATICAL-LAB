# Q29: Market Basket Analysis Using Apriori

import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori, association_rules

# 1. Load the transactional dataset (change the path to where your file is)
# Each row is one transaction; items are spread across columns (no header)
raw = pd.read_csv("/content/Market_Basket_Optimisation.csv", header=None, dtype=str)

print("Raw data shape:", raw.shape)

transactions = []
for _, row in raw.iterrows():
    items = [str(i).strip().lower() for i in row.dropna() if str(i).strip() != ""]
    transactions.append(sorted(set(items)))

print("Number of transactions:", len(transactions))
print("Average items per transaction:",
      round(sum(len(t) for t in transactions) / len(transactions), 2))
print("First 3 transactions:")
for t in transactions[:3]:
    print(t)

# 2. One-hot encode the transactions
encoder = TransactionEncoder()
onehot = encoder.fit(transactions).transform(transactions)
basket = pd.DataFrame(onehot, columns=encoder.columns_)

print("\nOne-hot encoded data shape:", basket.shape)
print("Unique items:", basket.shape[1])

# Most frequently bought items
print("\nTop 10 items by frequency:")
print(basket.mean().sort_values(ascending=False).head(10).round(4))

# 3. Apply Apriori with a minimum support threshold
min_support = 0.01
frequent_itemsets = apriori(basket, min_support=min_support, use_colnames=True)
frequent_itemsets["length"] = frequent_itemsets["itemsets"].apply(len)

print(f"\nFrequent itemsets found (min_support = {min_support}):",
      len(frequent_itemsets))
print(frequent_itemsets["length"].value_counts().sort_index()
      .rename("count").to_frame())

print("\nTop 10 frequent itemsets:")
print(frequent_itemsets.sort_values("support", ascending=False)
      .head(10).to_string(index=False))

# 4. Generate association rules with a minimum confidence threshold
min_confidence = 0.2
rules = association_rules(
    frequent_itemsets, metric="confidence", min_threshold=min_confidence
)

# Keep only useful rules (lift > 1 means a positive association)
rules = rules[rules["lift"] > 1]

# 5. Display support, confidence and lift
rules["antecedents"] = rules["antecedents"].apply(lambda x: ", ".join(sorted(x)))
rules["consequents"] = rules["consequents"].apply(lambda x: ", ".join(sorted(x)))

result = rules[["antecedents", "consequents", "support", "confidence", "lift"]]
result = result.sort_values("lift", ascending=False).round(4)

print(f"\nNumber of rules (min_confidence = {min_confidence}, lift > 1):",
      len(result))
print("\nTop 10 rules by lift:")
print(result.head(10).to_string(index=False))

print("\nTop 10 rules by confidence:")
print(result.sort_values("confidence", ascending=False)
      .head(10).to_string(index=False))
