# Q28: Evaluate K-Means Clustering

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score

# 1. Load the dataset
# changed: real file path (use "diabetes.csv" if the file is in your folder)
df = pd.read_csv("/content/diabetes.csv")

print("First five rows:")
print(df.head())

print("\nMissing values:")
print(df.isnull().sum())

# 2. Select numerical features
# Outcome is excluded because clustering is unsupervised.
features = [
    "Pregnancies", "Glucose", "BloodPressure",
    "SkinThickness", "Insulin", "BMI", "Pedigree", "Age"
]

X = df[features].copy()

# Treat biologically implausible zero measurements as missing.
# Zero pregnancies can be valid, so it is not replaced.
zero_columns = [
    "Glucose", "BloodPressure",
    "SkinThickness", "Insulin", "BMI"
]

X[zero_columns] = X[zero_columns].replace(0, float("nan"))

# 3. Impute missing values and standardize
imputer = SimpleImputer(strategy="median")
X_imputed = imputer.fit_transform(X)

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_imputed)

# 4. Fit K-Means for different K values
k_values = [2, 3, 4, 5, 6, 7]
inertias = []
silhouette_scores = []

for k in k_values:
    model = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )

    labels = model.fit_predict(X_scaled)

    inertias.append(model.inertia_)
    silhouette_scores.append(
        silhouette_score(X_scaled, labels)
    )

# 5. Display evaluation results
results = pd.DataFrame({
    "K": k_values,
    "Inertia": inertias,
    "Silhouette Score": silhouette_scores
})

print("\nK-Means evaluation:")
print(results.round(4))

# 6. Elbow plot
plt.plot(k_values, inertias, marker="o")
plt.xlabel("Number of clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method for Diabetes Dataset")
plt.xticks(k_values)
plt.grid(True)
plt.show()

# 7. Silhouette score plot
plt.plot(k_values, silhouette_scores, marker="o")
plt.xlabel("Number of clusters (K)")
plt.ylabel("Silhouette Score")
plt.title("Silhouette Score for Different K Values")
plt.xticks(k_values)
plt.grid(True)
plt.show()

# 8. Select K based on plots and evaluation
# This automatically chooses the K with the highest silhouette score.
best_k = k_values[silhouette_scores.index(max(silhouette_scores))]

print("\nSelected K based on silhouette score:", best_k)

final_model = KMeans(
    n_clusters=best_k,
    random_state=42,
    n_init=10
)

df["Cluster"] = final_model.fit_predict(X_scaled)

# 9. Count records in each cluster
print("\nRecords in each cluster:")
print(df["Cluster"].value_counts().sort_index())

# 10. Examine average feature values per cluster
profile_data = X.copy()
profile_data["Cluster"] = df["Cluster"]

print("\nCluster profiles:")
print(profile_data.groupby("Cluster").mean().round(2))

# 11. Visualize clusters in two dimensions using PCA
pca = PCA(n_components=2)
X_2d = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 6))

for cluster in sorted(df["Cluster"].unique()):
    mask = df["Cluster"].to_numpy() == cluster
    plt.scatter(
        X_2d[mask, 0],
        X_2d[mask, 1],
        label=f"Cluster {cluster}",
        alpha=0.7
    )

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("K-Means Clusters Visualized using PCA")
plt.legend()
plt.grid(True)
plt.show()

print(
    "\nVariance explained by two PCA components:",
    round(pca.explained_variance_ratio_.sum(), 4)
)
