# Q6: Examination Timetable Scheduling Using CSP

# Backtracking timetable generator
def solve_timetable(exams, slots, conflicts, timetable, index):
    if index == len(exams):
        return True

    exam = exams[index]

    for slot in slots:
        safe = True

        # Check conflicts with already scheduled exams
        for other_exam in conflicts[exam]:
            if timetable.get(other_exam) == slot:
                safe = False
                break

        if safe:
            timetable[exam] = slot

            if solve_timetable(
                exams, slots, conflicts, timetable, index + 1
            ):
                return True

            del timetable[exam]

    return False


# Main program
try:
    n = int(input("Enter number of examinations (minimum 6): "))
    if n < 6:
        print("Please enter at least 6 examinations.")
        raise SystemExit

    exams = []

    for i in range(n):
        exam = input(f"Enter examination {i + 1} name: ").strip()

        if not exam or exam in exams:
            print("Examination names must be non-empty and unique.")
            raise SystemExit

        exams.append(exam)

    number_of_slots = int(input("Enter number of time slots (minimum 4): "))

    if number_of_slots < 4:
        print("Please enter at least 4 time slots.")
        raise SystemExit

    slots = []

    for i in range(number_of_slots):
        slot = input(f"Enter time slot {i + 1}: ").strip()

        if not slot or slot in slots:
            print("Time slots must be non-empty and unique.")
            raise SystemExit

        slots.append(slot)

    # Collect student enrollment records
    students = int(input("Enter number of students: "))

    if students < 0:
        print("Number of students cannot be negative.")
        raise SystemExit

    enrollment = {}

    for i in range(students):
        student = input(f"Enter student {i + 1} ID: ").strip()

        if not student or student in enrollment:
            print("Student IDs must be non-empty and unique.")
            raise SystemExit

        print("Available exams:", ", ".join(exams))
        choices = input(
            "Enter exams this student takes, separated by commas: "
        ).split(",")

        choices = [x.strip() for x in choices if x.strip()]

        if any(x not in exams for x in choices):
            print("Invalid examination name in enrollment.")
            raise SystemExit

        enrollment[student] = list(set(choices))

    # Build examination conflict constraints
    conflicts = {exam: set() for exam in exams}

    for chosen_exams in enrollment.values():
        for exam1 in chosen_exams:
            for exam2 in chosen_exams:
                if exam1 != exam2:
                    conflicts[exam1].add(exam2)

    # Order exams by number of conflicts (most constrained first)
    exams.sort(key=lambda exam: len(conflicts[exam]), reverse=True)

    timetable = {}

    if solve_timetable(exams, slots, conflicts, timetable, 0):
        print("\n===== EXAMINATION TIMETABLE =====")

        for slot in slots:
            print(f"\nTime slot: {slot}")

            scheduled = [
                exam for exam in exams if timetable[exam] == slot
            ]

            if scheduled:
                for exam in scheduled:
                    print("  ", exam)
            else:
                print("   No examinations")

        # Verify that conflicting exams have different slots
        valid = True

        for exam in exams:
            for other_exam in conflicts[exam]:
                if timetable[exam] == timetable[other_exam]:
                    valid = False

        if valid:
            print("\nValidation: No conflicting examinations overlap.")
        else:
            print("\nValidation failed: Timetable has conflicts.")

        print("Valid timetable found: Yes")
    else:
        print("\nNo valid timetable exists with the available time slots.")
        print("The available time slots are insufficient for this schedule.")

except ValueError:
