# Q7: Tic-Tac-Toe using Minimax Algorithm

# Display the game board
def print_board(board):
    print()
    for i in range(0, 9, 3):
        print(" | ".join(board[i:i+3]))
        if i < 6:
            print("---------")
    print()


# Check winner
def check_winner(board):
    wins = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),
        (0, 3, 6), (1, 4, 7), (2, 5, 8),
        (0, 4, 8), (2, 4, 6)
    ]

    for a, b, c in wins:
        if board[a] == board[b] == board[c] != " ":
            return board[a]

    return None


# Minimax algorithm
def minimax(board, is_maximizing):
    winner = check_winner(board)

    if winner == "O":
        return 1, None

    if winner == "X":
        return -1, None

    if " " not in board:
        return 0, None

    if is_maximizing:
        best_score = -float("inf")
        best_move = None

        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                score, _ = minimax(board, False)
                board[i] = " "

                if score > best_score:
                    best_score = score
                    best_move = i

        return best_score, best_move

    else:
        best_score = float("inf")
        best_move = None

        for i in range(9):
            if board[i] == " ":
                board[i] = "X"
                score, _ = minimax(board, True)
                board[i] = " "

                if score < best_score:
                    best_score = score
                    best_move = i

        return best_score, best_move


# Play the game
def play_game():
    board = [" "] * 9

    print("===== TIC-TAC-TOE =====")
    print("You are X. Computer is O.")
    print("Enter positions from 1 to 9.")

    while True:
        # Show position numbers for empty cells
        display = [
            str(i + 1) if board[i] == " " else board[i]
            for i in range(9)
        ]
        print_board(display)

        # Human move
        try:
            move = int(input("Enter your move (1-9): ")) - 1
        except ValueError:
            print("Invalid input! Enter a number from 1 to 9.")
            continue

        if move < 0 or move > 8 or board[move] != " ":
            print("Invalid move! Choose an empty position.")
            continue

        board[move] = "X"

        if check_winner(board) == "X":
            print_board(board)
            print("Congratulations! You win.")
            break

        if " " not in board:
            print_board(board)
            print("Game is a draw!")
            break

        # Computer move using Minimax
        score, computer_move = minimax(board, True)
        board[computer_move] = "O"

        print("Computer chooses position:", computer_move + 1)
        print_board(board)

        if check_winner(board) == "O":
            print("Computer wins!")
            break

        if " " not in board:
            print("Game is a draw!")
            break


# Start the game
play_game()
