import heapq

# Print the puzzle board
def print_board(state):
    for i in range(0, 9, 3):
        print(" | ".join("_" if x == 0 else str(x)
                         for x in state[i:i+3]))
    print()


# Manhattan distance heuristic
def heuristic(state, goal):
    distance = 0

    for i in range(9):
        tile = state[i]

        if tile != 0:
            goal_pos = goal.index(tile)

            row_distance = abs(i // 3 - goal_pos // 3)
            col_distance = abs(i % 3 - goal_pos % 3)

            distance += row_distance + col_distance

    return distance


# Generate valid moves of the empty tile (0)
def get_neighbours(state):
    neighbours = []
    zero = state.index(0)

    row = zero // 3
    col = zero % 3

    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    for dr, dc in moves:
        new_row = row + dr
        new_col = col + dc

        if 0 <= new_row < 3 and 0 <= new_col < 3:
            new_index = new_row * 3 + new_col

            new_state = list(state)
            new_state[zero], new_state[new_index] = (
                new_state[new_index], new_state[zero]
            )

            neighbours.append(tuple(new_state))

    return neighbours


# Check whether a puzzle is solvable
def is_solvable(start, goal):
    def inversions(state):
        tiles = [x for x in state if x != 0]
        count = 0

        for i in range(len(tiles)):
            for j in range(i + 1, len(tiles)):
                if tiles[i] > tiles[j]:
                    count += 1

        return count

    return inversions(start) % 2 == inversions(goal) % 2


# A* search algorithm
def astar(start, goal):
    priority_queue = [(heuristic(start, goal), 0, start)]

    g_cost = {start: 0}
    parent = {start: None}
    explored = set()

    while priority_queue:
        f, cost, current = heapq.heappop(priority_queue)

        # Ignore outdated or repeated states
        if current in explored:
            continue

        explored.add(current)

        # Goal reached
        if current == goal:
            path = []
            node = current

            while node is not None:
                path.append(node)
                node = parent[node]

            path.reverse()
            return path, len(explored)

        # Explore neighbouring states
        for neighbour in get_neighbours(current):
            new_cost = g_cost[current] + 1

            if new_cost < g_cost.get(neighbour, float("inf")):
                g_cost[neighbour] = new_cost
                parent[neighbour] = current

                f_cost = new_cost + heuristic(neighbour, goal)

                heapq.heappush(
                    priority_queue,
                    (f_cost, new_cost, neighbour)
                )

    return None, len(explored)

        else:

except ValueError:
