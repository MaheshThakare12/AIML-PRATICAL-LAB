# Q27: Mall Customer Clustering using K-Means

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# 1. Load dataset
# changed: real file path (use "Mall_Customers.csv" if the file is in your folder)
df = pd.read_csv("/content/Mall_Customers.csv")

print("Dataset shape:", df.shape)
print(df.head())
print("\nMissing values:\n", df.isnull().sum())

# 2. Select annual income and spending score
features = ["Annual Income (k$)", "Spending Score (1-100)"]
X = df[features].copy()
X = X.fillna(X.median())

# 3. Standardize the features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 4. Elbow method to choose K
inertia = []

for k in range(1, 11):
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    km.fit(X_scaled)
    inertia.append(km.inertia_)

plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("K (Number of clusters)")
plt.ylabel("Inertia")
plt.title("Elbow Method for Mall Customers")
plt.xticks(range(1, 11))
plt.grid(True)
plt.show()

# 5. Fit K-Means
# Five clusters: the elbow bends at K=5.
k = 5

kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
df["Cluster"] = kmeans.fit_predict(X_scaled)

# Convert centroid coordinates back to original units
centroids = scaler.inverse_transform(kmeans.cluster_centers_)

# 6. Count customers in each cluster
counts = df["Cluster"].value_counts().sort_index()

print("\nNumber of customers in each cluster:")
print(counts)

# 7. Visualize clusters and centroids
plt.figure(figsize=(9, 6))

for cluster in range(k):
    group = df[df["Cluster"] == cluster]
    plt.scatter(
        group["Annual Income (k$)"],
        group["Spending Score (1-100)"],
        label=f"Cluster {cluster}"
    )

plt.scatter(
    centroids[:, 0],
    centroids[:, 1],
    marker="X",
    s=250,
    edgecolor="black",
    label="Centroids"
)

plt.xlabel("Annual Income (k$)")
plt.ylabel("Spending Score (1-100)")
plt.title("Mall Customer Clusters")
plt.legend()
plt.grid(True)
plt.show()

# 8. Print average feature values for each cluster
print("\nCluster profiles:")
print(df.groupby("Cluster")[features].mean().round(2))
