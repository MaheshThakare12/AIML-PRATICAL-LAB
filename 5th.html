# Q5: Sudoku Solver Using CSP and Backtracking

# Display the Sudoku grid
def print_grid(grid):
    for i in range(9):
        if i % 3 == 0 and i != 0:
            print("- - - - - - - - - - -")

        for j in range(9):
            if j % 3 == 0 and j != 0:
                print("|", end=" ")

            print(grid[i][j], end=" ")

        print()


# Check whether a number can be placed
def is_valid(grid, row, col, num):
    # Check row
    for j in range(9):
        if grid[row][j] == num:
            return False

    # Check column
    for i in range(9):
        if grid[i][col] == num:
            return False

    # Check 3x3 subgrid
    start_row = (row // 3) * 3
    start_col = (col // 3) * 3

    for i in range(start_row, start_row + 3):
        for j in range(start_col, start_col + 3):
            if grid[i][j] == num:
                return False

    return True


# Solve Sudoku using backtracking
def solve_sudoku(grid):
    for row in range(9):
        for col in range(9):
            if grid[row][col] == 0:
                for num in range(1, 10):
                    if is_valid(grid, row, col, num):
                        grid[row][col] = num

                        if solve_sudoku(grid):
                            return True

                        grid[row][col] = 0

                return False

    return True


# Validate completed Sudoku
def validate_grid(grid):
    required = set(range(1, 10))

    for i in range(9):
        if set(grid[i]) != required:
            return False

        if {grid[r][i] for r in range(9)} != required:
            return False

    for row in range(0, 9, 3):
        for col in range(0, 9, 3):
            values = set()

            for i in range(row, row + 3):
                for j in range(col, col + 3):
                    values.add(grid[i][j])

            if values != required:
                return False

    return True


# Main program
print("Enter Sudoku grid row by row.")
print("Use 0 for empty cells.")

grid = []

try:
    for i in range(9):
        row = list(map(int, input(
            f"Enter row {i + 1} (9 numbers separated by spaces): "
        ).split()))

        if len(row) != 9 or any(num < 0 or num > 9 for num in row):
            raise ValueError

        grid.append(row)

    # Check whether the initial clues contain conflicts
    valid_input = True

    for row in range(9):
        for col in range(9):
            num = grid[row][col]

            if num != 0:
                grid[row][col] = 0

                if not is_valid(grid, row, col, num):
                    valid_input = False

                grid[row][col] = num

    if not valid_input:
        print("Invalid puzzle: conflicting initial values.")
    elif solve_sudoku(grid):
        print("\nCompleted Sudoku:")
        print_grid(grid)

        if validate_grid(grid):
            print("Validation: Sudoku solution is valid.")
            print("Solution found: Yes")
        else:
            print("Validation failed.")
    else:
        print("Unsolvable Sudoku puzzle.")
        print("Solution found: No")

except ValueError:
    print("Invalid input! Enter 9 rows with 9 integers (0 to 9).")
