# Q1: A* Search Algorithm for Route Finding

import heapq
import math

# 1. Define city coordinates
cities = {
    "Pune": (0, 0),
    "Mumbai": (2, 0),
    "Nashik": (2, 2),
    "Surat": (4, 0),
    "Indore": (4, 3),
    "Vadodara": (6, 1),
    "Ahmedabad": (7, 2),
    "Nagpur": (7, 5)
}

# 2. Define weighted graph
graph = {
    "Pune": [("Mumbai", 2), ("Nashik", 3)],
    "Mumbai": [("Pune", 2), ("Surat", 3), ("Nashik", 2)],
    "Nashik": [("Pune", 3), ("Mumbai", 2), ("Indore", 4)],
    "Surat": [("Mumbai", 3), ("Vadodara", 2)],
    "Indore": [("Nashik", 4), ("Vadodara", 3), ("Nagpur", 4)],
    "Vadodara": [("Surat", 2), ("Indore", 3), ("Ahmedabad", 2)],
    "Ahmedabad": [("Vadodara", 2), ("Nagpur", 4)],
    "Nagpur": [("Indore", 4), ("Ahmedabad", 4)]
}

# 3. Heuristic function: straight-line distance
def heuristic(city, goal):
    x1, y1 = cities[city]
    x2, y2 = cities[goal]
    return math.sqrt((x1 - x2)**2 + (y1 - y2)**2)

# 4. A* search function
def astar(start, goal):
    # Priority queue stores (f_cost, g_cost, city)
    pq = [(heuristic(start, goal), 0, start)]

    g_cost = {start: 0}
    parent = {start: None}
    explored = set()

    while pq:
        f, cost, current = heapq.heappop(pq)

        if current in explored:
            continue

        explored.add(current)

        # Goal reached
        if current == goal:
            path = []
            node = goal

            while node is not None:
                path.append(node)
                node = parent[node]

            path.reverse()
            return path, g_cost[goal], len(explored)

        # Explore neighboring cities
        for neighbor, distance in graph[current]:
            new_cost = g_cost[current] + distance

            if new_cost < g_cost.get(neighbor, float("inf")):
                g_cost[neighbor] = new_cost
                parent[neighbor] = current

                f_cost = new_cost + heuristic(neighbor, goal)
                heapq.heappush(pq, (f_cost, new_cost, neighbor))

    return None, 0, len(explored)

# 5. Accept source and destination from user
print("Available cities:", ", ".join(cities.keys()))

start = input("Enter source city: ").strip().title()
goal = input("Enter destination city: ").strip().title()

# 6. Display the result
if start not in cities or goal not in cities:
    print("Invalid city name!")

else:
    path, cost, explored = astar(start, goal)

    if path:
        print("Selected route:", " -> ".join(path))
        print("Total path cost:", cost)
        print("Nodes explored:", explored)
    else:
        print("No route found.")
