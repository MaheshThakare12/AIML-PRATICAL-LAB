# Q3: Robot Navigation using A* Search

import heapq

# 1. Create an 8 x 8 grid
SIZE = 8

# 2. Manhattan distance heuristic
def heuristic(cell, goal):
    return abs(cell[0] - goal[0]) + abs(cell[1] - goal[1])

# 3. A* Search algorithm
def astar(start, goal, obstacles):
    pq = [(heuristic(start, goal), 0, start)]

    parent = {start: None}
    g_cost = {start: 0}
    explored = set()

    # Up, down, left, right
    directions = [(-1, 0), (1, 0), (0, -1), (0, 1)]

    while pq:
        f, cost, current = heapq.heappop(pq)

        if current in explored:
            continue

        explored.add(current)

        # Goal reached
        if current == goal:
            path = []
            node = goal

            while node is not None:
                path.append(node)
                node = parent[node]

            path.reverse()
            return path, len(path) - 1, len(explored)

        # Check neighboring cells
        for dr, dc in directions:
            nr = current[0] + dr
            nc = current[1] + dc
            neighbor = (nr, nc)

            # Check grid boundaries and obstacles
            if not (0 <= nr < SIZE and 0 <= nc < SIZE):
                continue

            if neighbor in obstacles or neighbor in explored:
                continue

            new_cost = g_cost[current] + 1

            if new_cost < g_cost.get(neighbor, float("inf")):
                g_cost[neighbor] = new_cost
                parent[neighbor] = current

                f_cost = new_cost + heuristic(neighbor, goal)
                heapq.heappush(
                    pq, (f_cost, new_cost, neighbor)
                )

    return None, 0, len(explored)

# 4. Display the grid
def display_grid(path, start, goal, obstacles):
    path_cells = set(path) if path else set()

    print("\nRobot Navigation Grid:")
    print("S = Start, G = Goal, # = Obstacle")
    print("* = Path, . = Free cell\n")

    for r in range(SIZE):
        for c in range(SIZE):
            cell = (r, c)

            if cell == start:
                symbol = "S"
            elif cell == goal:
                symbol = "G"
            elif cell in obstacles:
                symbol = "#"
            elif cell in path_cells:
                symbol = "*"
            else:
                symbol = "."

            print(symbol, end=" ")
        print()

# 5. Take input from the user
print("Grid coordinates range from 0 to 7.")
print("Enter coordinates as: row column")

start = tuple(map(int, input("Enter start cell: ").split()))
goal = tuple(map(int, input("Enter goal cell: ").split()))

# Validate start and goal
if len(start) != 2 or len(goal) != 2:
    print("Invalid coordinates. Enter row and column.")
    raise SystemExit

def valid_cell(cell):
    return (
        len(cell) == 2
        and 0 <= cell[0] < SIZE
        and 0 <= cell[1] < SIZE
    )

if not valid_cell(start) or not valid_cell(goal):
    print("Invalid cell! Row and column must be between 0 and 7.")
    raise SystemExit

if start == goal:
    print("Start and goal must be different.")
    raise SystemExit

# Enter obstacle locations
obstacles = set()
n = int(input("Enter number of obstacles: "))

for i in range(n):
    cell = tuple(map(int, input(
        f"Enter obstacle {i + 1} (row column): "
    ).split()))

    if not valid_cell(cell):
        print("Invalid obstacle coordinates!")
        raise SystemExit

    if cell == start or cell == goal:
        print("Obstacle cannot be placed on start or goal!")
        raise SystemExit

    obstacles.add(cell)

# 6. Run A* and display results
path, length, explored = astar(start, goal, obstacles)

display_grid(path, start, goal, obstacles)

if path:
    print("\nSelected path:", path)
