# Q2: Greedy Best-First Search Algorithm

import heapq
import math

# 1. Define city coordinates for heuristic calculation
cities = {
    "Pune": (0, 0),
    "Mumbai": (2, 0),
    "Nashik": (2, 2),
    "Surat": (4, 0),
    "Indore": (4, 3),
    "Vadodara": (6, 1),
    "Ahmedabad": (7, 2),
    "Nagpur": (7, 5)
}

# 2. Define weighted graph
graph = {
    "Pune": [("Mumbai", 2), ("Nashik", 3)],
    "Mumbai": [("Pune", 2), ("Surat", 3), ("Nashik", 2)],
    "Nashik": [("Pune", 3), ("Mumbai", 2), ("Indore", 4)],
    "Surat": [("Mumbai", 3), ("Vadodara", 2)],
    "Indore": [("Nashik", 4), ("Vadodara", 3), ("Nagpur", 4)],
    "Vadodara": [("Surat", 2), ("Indore", 3), ("Ahmedabad", 2)],
    "Ahmedabad": [("Vadodara", 2), ("Nagpur", 4)],
    "Nagpur": [("Indore", 4), ("Ahmedabad", 4)]
}

# 3. Heuristic function: estimated distance to destination
def heuristic(city, goal):
    x1, y1 = cities[city]
    x2, y2 = cities[goal]
    return math.sqrt((x2 - x1)**2 + (y2 - y1)**2)

# 4. Greedy Best-First Search
def greedy_best_first(start, goal):
    pq = [(heuristic(start, goal), start)]
    parent = {start: None}
    total_cost = {start: 0}
    visited = set()

    while pq:
        h, current = heapq.heappop(pq)

        if current in visited:
            continue

        visited.add(current)

        # Destination reached
        if current == goal:
            path = []
            node = goal

            while node is not None:
                path.append(node)
                node = parent[node]

            path.reverse()
            return path, total_cost[goal], len(visited)

        # Explore neighboring cities
        for neighbor, distance in graph[current]:
            if neighbor not in visited and neighbor not in parent:
                parent[neighbor] = current
                total_cost[neighbor] = total_cost[current] + distance
                heapq.heappush(
                    pq, (heuristic(neighbor, goal), neighbor)
                )

    return None, 0, len(visited)

# 5. Take input from user
print("Available cities:", ", ".join(cities.keys()))

start = input("Enter source city: ").strip().title()
goal = input("Enter destination city: ").strip().title()

# 6. Display result
if start not in cities or goal not in cities:
    print("Invalid city name!")

else:
    path, cost, explored = greedy_best_first(start, goal)

    if path:
        print("Selected route:", " -> ".join(path))
        print("Total path cost:", cost)
        print("Nodes explored:", explored)
    else:
        print("No route found.")
