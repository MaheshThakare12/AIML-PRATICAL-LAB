# Q26: Student Grouping using K-Means

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score      # added

# 1. Load student dataset
# changed: real file path (use "Student_Performance.csv" if the file is in your folder)
df = pd.read_csv("/content/Student_Performance.csv")

print(df.head())
print("\nMissing values:\n", df.isnull().sum())

# 2. Select at least three academic-related features
features = [
    "Hours Studied",
    "Previous Scores",
    "Sample Question Papers Practiced"
]

X = df[features].copy()

# 3. Impute missing values and standardize
imputer = SimpleImputer(strategy="median")
X_imputed = imputer.fit_transform(X)

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_imputed)

# 4. Elbow method
inertias = []

for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X_scaled)
    inertias.append(model.inertia_)

plt.plot(range(1, 11), inertias, marker="o")
plt.xlabel("Number of clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method for Student Grouping")
plt.xticks(range(1, 11))
plt.grid(True)
plt.show()

# 5. Fit K-Means
# Choose K after inspecting the elbow plot.
k = 3

kmeans = KMeans(n_clusters=k, random_state=42, n_init=10)
df["Cluster"] = kmeans.fit_predict(X_scaled)

# added: silhouette score to check how well separated the clusters are
print("\nSilhouette score for K =", k, ":",
      round(silhouette_score(X_scaled, df["Cluster"]), 4))

print("\nStudents per cluster:")
print(df["Cluster"].value_counts().sort_index())

# 6. Cluster profiles in original feature units
print("\nCluster profiles:")
print(df.groupby("Cluster")[features].mean().round(2))

# 7. Reduce features to two dimensions using PCA
pca = PCA(n_components=2)
X_2d = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 5))

for cluster in sorted(df["Cluster"].unique()):
    mask = df["Cluster"].to_numpy() == cluster
    plt.scatter(
        X_2d[mask, 0],
        X_2d[mask, 1],
        label=f"Cluster {cluster}"
    )

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("Student Clusters Visualized using PCA")
plt.legend()
plt.grid(True)
plt.show()

print(
    "\nVariance explained by the two PCA components:",
    round(pca.explained_variance_ratio_.sum(), 4)
)
